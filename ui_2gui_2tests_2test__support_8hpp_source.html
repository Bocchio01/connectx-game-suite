68 bool start(quint16 preferredPort = 0)
69 {
70 stop();
71
72 for (int attempt = 0; attempt < 3; ++attempt)
73 {
74 const quint16 candidate = (attempt == 0 && preferredPort != 0) ? preferredPort : freePort();
75 if (candidate == 0)
76 {
77 continue;
78 }
79
80 process_ = std::make_unique<QProcess>();
81 process_->setWorkingDirectory(workDir_.path());
82 process_->setProcessChannelMode(QProcess::MergedChannels);
83 process_->start(serverExecutable(), {QStringLiteral("--port"), QString::number(candidate)});
84 // (the first launch of a freshly built executable can be slow: virus scanners look at it)
85 if (!process_->waitForStarted(30000))
86 {
87 continue;
88 }
89
90 // Wait for the banner: the server is listening from then on.
91 output_.clear();
92 QElapsedTimer clock;
93 clock.start();
94 while (clock.elapsed() < 30000)
95 {
96 process_->waitForReadyRead(100);
97 output_ += QString::fromLocal8Bit(process_->readAll());
98 if (output_.contains(QStringLiteral("Server started successfully")))
99 {
100 port_ = candidate;
101 return true;
102 }
103 if (process_->state() == QProcess::NotRunning)
104 {
105 break; // e.g. the port was taken in the meantime: try another one
106 }
107 }
108 stop();
109 }
110 return false;
111 }
112
113 void stop()
114 {
115 if (process_ && process_->state() != QProcess::NotRunning)
116 {
117 process_->kill();
118 process_->waitForFinished(3000);
119 }
120 process_.reset();
121 port_ = 0;
122 }
123
124 bool isRunning() const { return process_ && process_->state() == QProcess::Running; }
125 quint16 port() const { return port_; }
126 QString output() const { return output_; }

128 private:

131 quint16 port_ = 0;
